! R1 Minimal Base Configuration
! IP: 192.168.122.254 | User: admin | Pass: cisco
! Gi0/0=LAN/Internet, Gi0/1=R2, Gi0/2=R3

enable
configure terminal
hostname R1
no ip domain-lookup

enable secret cisco
username admin privilege 15 secret cisco

! SSH Setup (RSA KEYS REQUIRED)
ip domain-name lab.local
crypto key generate rsa modulus 1024
ip ssh version 2

! Interfaces
interface GigabitEthernet0/0
 description LAN_Internet
 ip address 192.168.122.254 255.255.255.0
 no shutdown
exit

interface GigabitEthernet0/1
 description Link_to_R2
 ip address 10.0.12.1 255.255.255.252
 no shutdown
exit

interface GigabitEthernet0/2
 description Link_to_R3
 ip address 10.0.13.1 255.255.255.252
 no shutdown
exit

! Static routes to reach all networks
ip route 10.0.20.0 255.255.255.0 10.0.12.2
ip route 10.0.30.0 255.255.255.0 10.0.13.2

! VTY lines (SSH access)
line vty 0 4
 login local
 transport input ssh
 exec-timeout 30 0
exit

end
write memory
