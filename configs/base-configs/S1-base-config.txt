! S1 Minimal Base Configuration - SSH + SVI + Default Gateway
!
enable
configure terminal
!
no ip domain-lookup
hostname S1
!
! Disable L3 routing (use ip default-gateway instead)
no ip routing
!
! AAA
username admin privilege 15 secret cisco
!
! Domain name for SSH keys
ip domain-name lab.local
!
! Generate RSA keys for SSH (REQUIRED for SSH)
crypto key generate rsa modulus 2048
!
! Access port to R2 (active in VLAN 1)
interface GigabitEthernet0/0
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit
!
! Management SVI (MUST BE UP)
interface Vlan1
 ip address 10.0.20.10 255.255.255.0
 no shutdown
 exit
!
! Default Gateway
ip default-gateway 10.0.20.1

! SSH Configuration
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
 exec-timeout 30 0
 exit
!
end
write memory
