! R4 Test Router - Direct Ethernet Connection
! IP: 192.168.122.100 | User: admin | Pass: cisco
! Gi0/0=Direct to Ethernet Switch

enable
configure terminal
hostname R4
no ip domain-lookup

enable secret cisco
username admin privilege 15 secret cisco

! SSH Setup (RSA KEYS REQUIRED)
ip domain-name lab.local
crypto key generate rsa modulus 2048
ip ssh version 2

! Interfaces
interface GigabitEthernet0/0
 description Direct_to_Switch
 ip address 192.168.122.100 255.255.255.0
 no shutdown
exit

! VTY lines (SSH access)
line vty 0 4
 login local
 transport input ssh
 exec-timeout 30 0
exit

end
write memory
