! R2 Minimal Base Configuration
! IP: 10.0.12.2 | User: admin | Pass: cisco
! Gi0/0=R1, Gi0/1=S1

enable
configure terminal
hostname R2
no ip domain-lookup

enable secret cisco
username admin privilege 15 secret cisco

! SSH Setup (RSA KEYS REQUIRED)
ip domain-name lab.local
crypto key generate rsa modulus 1024
ip ssh version 2

! Interfaces
interface GigabitEthernet0/0
 description Link_to_R1
 ip address 10.0.12.2 255.255.255.252
 no shutdown
exit

interface GigabitEthernet0/1
 description LAN_to_S1
 ip address 10.0.20.1 255.255.255.0
 no shutdown
exit

! Default route to R1
ip route 0.0.0.0 0.0.0.0 10.0.12.1

! VTY lines (SSH access)
line vty 0 4
 login local
 transport input ssh
 exec-timeout 30 0
exit

end
write memory
